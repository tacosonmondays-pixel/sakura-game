async function setup(vp) {
  const [w, h] = vp.split('x').map(Number);
  const context = await browser.newContext({ viewport: { width: w, height: h }, deviceScaleFactor: 1, hasTouch: true });
  const page = await context.newPage();
  const errors = []; const net = [];
  page.on('pageerror', (e) => errors.push(e.message));
  page.on('request', (r) => { const u = r.url(); if (!u.startsWith(base) && !u.startsWith('data:')) net.push(`${r.method()} ${u}`); });
  await page.goto(`${base}#/lobby`, { waitUntil: 'load' });
  await page.evaluate(() => { localStorage.clear(); sessionStorage.clear(); });
  await page.goto(`${base}#/lobby`, { waitUntil: 'load' });
  const later = await page.waitForSelector('[data-testid="beta-gift-later"]', { timeout: 30000 }).catch(() => null);
  if (later) await later.click({ force: true }).catch(() => {});
  await page.waitForFunction(() => window.__sakura?.store?.profile, null, { timeout: 60000 });
  await page.evaluate(() => document.querySelectorAll('.modal-overlay').forEach((m) => m.remove()));
  await page.evaluate(async () => {
    const { store } = window.__sakura;
    const { CAMPAIGN_IDS } = await import('/src/data/stages.js');
    for (const id of CAMPAIGN_IDS.slice(0, 12)) store.profile.progress.stages[id] = { easy: false, normal: true, hard: false, nightmare: false, clears: 1 };
    store.commit('replace'); store.saveNow?.();
    window.__commits = [];
    const c = store.commit.bind(store); store.commit = (r, o) => { window.__commits.push(r); return c(r, o); };
    const s = store.saveNow?.bind(store); if (s) store.saveNow = () => { window.__commits.push('saveNow'); return s(); };
  });
  await page.waitForTimeout(1500);
  return { context, page, errors, net };
}
const shot = async (page, name) => {
  for (const q of [70, 50, 35]) { const path = `${SHOTS}/${name}.jpg`; await page.screenshot({ path, type: 'jpeg', quality: q }); if (statSync(path).size <= 300 * 1024) return; }
};
const overflow = (page) => page.evaluate(() => {
  const bad = [];
  if (document.documentElement.scrollWidth > innerWidth + 1) bad.push('doc ' + document.documentElement.scrollWidth);
  for (const el of document.querySelectorAll('.sbx-tabs button, .sbx-chip, .sbx-build select, .sbx-field input, [data-testid^="sandbox-"]')) {
    const r = el.getBoundingClientRect(); if (!r.width) continue;
    if (r.right > innerWidth + 1 || r.left < -1) bad.push(`${el.dataset.testid || el.className} x ${Math.round(r.left)}..${Math.round(r.right)}`);
  }
  return bad;
});
const small = (page) => page.evaluate(() => [...document.querySelectorAll('button, select, input')].filter((e) => e.closest('.sbx-root, [data-testid="sandbox-maps"], [data-testid="sandbox-roster"], .sbx-girl-modal') || /sandbox/.test(e.dataset.testid || '')).map((e) => [e.dataset.testid || e.textContent.trim().slice(0, 14), e.getBoundingClientRect()]).filter(([, r]) => r.width && (r.height < 32 || r.width < 32)).map(([n, r]) => `${n} ${Math.round(r.width)}x${Math.round(r.height)}`));

async function battle(page, hash, label, act, expect) {
  await page.evaluate(() => { location.hash = '#/sandbox'; }); await page.waitForSelector('[data-testid="sandbox-maps"]', { timeout: 60000 }); await page.waitForTimeout(800);
  await page.evaluate(() => { window.__battle = null; });
  const before = await page.evaluate(ALL); const pBefore = await page.evaluate(P);
  await page.evaluate(() => { window.__commits = []; });
  await page.evaluate((x) => { location.hash = x; }, hash);
  const ok = await page.waitForFunction((x) => window.__battle?.ui?.sandbox && window.__battle.sim?.stageId === x, expect, { timeout: 90000 }).then(() => true).catch(() => false);
  check(ok, `${label}: battle mounted`);
  if (!ok) return;
  const info = await page.evaluate(() => { const { sim, ui } = window.__battle; return { stage: sim.stageId, diff: sim.difficulty, fp: !!sim.freeplay, wave: sim.wave, titanReq: !!ui.req.titan, fpReq: ui.req.freeplay, saver: !!ui.saver }; });
  console.log('   ', label, JSON.stringify(info));
  check(!info.titanReq && !info.fpReq && !info.saver, `${label}: no req.titan / req.freeplay / saver`);
  await act();
  await page.waitForTimeout(1500);
  const commits = await page.evaluate(() => window.__commits);
  check(commits.length === 0, `${label}: no store.commit/saveNow (${commits.join(',')})`);
  check((await page.evaluate(ALL)) === before, `${label}: localStorage byte-identical`);
  check((await page.evaluate(P)) === pBefore, `${label}: in-memory profile identical`);
}
