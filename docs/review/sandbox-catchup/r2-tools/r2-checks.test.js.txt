import { parseBattleParams, createBattleSim } from '../../src/ui/battle/setup.js';

import { CAMPAIGN_IDS } from '../../src/data/stages.js';
import { UNITS } from '../../src/data/units.js';

import { gainHeroXp, heroXpCap } from '../../src/sim/hero.js';

const unlocked = () => { const p = createProfile(); for (const id of CAMPAIGN_IDS.slice(0, 7)) p.progress.stages[id] = { easy: true, normal: false, hard: false, nightmare: false, clears: 1 }; return p; };
afterEach(() => vi.useRealTimers());

describe('r2 typical roll', () => {
  it('typical < max for every line, every unit, every preset', () => {
    let lines = 0, eq = [];
    for (const u of UNITS) for (const preset of SANDBOX_GEAR_PRESETS) {
      const mx = sandboxGear(u.id, preset, 'max'), ty = sandboxGear(u.id, preset, 'typical');
      expect(ty.length).toBe(mx.length);
      mx.forEach((g, i) => {
        const a = gearLines(g), b = gearLines(ty[i]);
        const pairs = [[a.main, b.main], ...a.subs.map((s, j) => [s, b.subs[j]])];
        for (const [m, t] of pairs) { lines++; expect(t.stat).toBe(m.stat); if (!(t.value < m.value)) eq.push(`${u.id}/${preset}/${g.slot}/${m.stat} ${t.value} vs ${m.value}`); }
      });
    }
    console.log('lines', lines, 'not-less', eq.length, eq.slice(0, 10));
    expect(eq).toEqual([]);
  });
  it('typical base = Monte Carlo mean of real legendary rolls', () => {
    const rng = createRng('rv');
    const acc = {};
    for (let i = 0; i < 40000; i++) {
      const g = rollGear(rng, { rarity: 'legendary', slot: ['weapon','armor','charm'][i % 3] });
      (acc['m:' + g.main.stat] ||= []).push(g.main.base);
      for (const s of g.subs) (acc['s:' + s.stat] ||= []).push(s.base);
    }
    const out = [];
    for (const [k, v] of Object.entries(acc)) {
      const mean = v.reduce((a, b) => a + b, 0) / v.length;
      const [kind, stat] = k.split(':');
      const r = kind === 'm' ? mainStatRange(stat, 'legendary') : substatRange(stat, 'legendary');
      const mid = (r[0] + r[1]) / 2;
      out.push([k, v.length, +mean.toFixed(4), mid, r, +(Math.abs(mean - mid) / (r[1] - r[0] || 1)).toFixed(4)]);
    }
    console.log(JSON.stringify(out));
    for (const o of out) expect(o[5]).toBeLessThan(0.03);
  });
});

describe('r2 resonance', () => {
  const inWeek = gameDayStart(RESONANCE_FROM_DAY) + 3 * 86400e3;
  const before = gameDayStart(RESONANCE_FROM_DAY) - 3 * 86400e3;
  it('on/off == the real clock in/out of a resonance week, every unit, several builds', () => {
    const builds = [SANDBOX_MAX_BUILD, normalizeBuild({ level: 60, stars: 4, lr: false, gear: 'leg' }), normalizeBuild({ stars: 5, lr: false, gear: 'none' }), normalizeBuild({ roll: 'typical' })];
    let n = 0; const bad = [];
    for (const [mode, t] of [['on', inWeek], ['off', before]]) {
      for (const u of UNITS) for (const b of builds) {
        vi.useFakeTimers(); vi.setSystemTime(t);
        expect(resonanceOn()).toBe(mode === 'on');
        const clock = sandboxUnitStats(u.id, b);
        vi.useRealTimers();
        vi.useFakeTimers(); vi.setSystemTime(mode === 'on' ? before : inWeek);
        const forced = sandboxUnitStats(u.id, b, { resonance: mode });
        vi.useRealTimers();
        n++;
        for (const k of Object.keys(clock)) if (JSON.stringify(clock[k]) !== JSON.stringify(forced[k])) bad.push(`${mode} ${u.id} L${b.level} ${k}: clock ${clock[k]} forced ${forced[k]}`);
      }
    }
    console.log('resonance cases', n, 'diffs', bad.length, bad.slice(0, 12));
    expect(bad).toEqual([]);
  });
});

describe('r2 hero', () => {
  const mk = (params, profile = unlocked()) => {
    const req = parseBattleParams({ stage: '4-5', team: 'kage', hero: 'hikari', ...params }, profile);
    return { sim: createBattleSim(profile, { ...req, seed: 4 }).sim, run: req.sandboxRun, req };
  };
  const place = (s) => { for (let y = 1; y < s.map.height; y++) for (let x = 1; x < s.map.width; x++) { const t = s.placeTower('hikari', x + 0.5, y + 0.5); if (t) return t; } return null; };
  const snap = (t) => Object.fromEntries(Object.entries(t).filter(([k, v]) => typeof v === 'number' || typeof v === 'boolean').sort());
  it('late-start Lv25+Mastery equals a normally levelled hero; no hook leak into a normal battle', () => {
    const a = mk({ sandbox: '1', wave: '200' });
    const da = createSandboxDriver(a.sim, { heroMax: () => a.run.heroMax && a.run.startWave > 1 });
    const ha = place(a.sim);
    const b = mk({ sandbox: '1', wave: '200', hmax: '0' });
    const db = createSandboxDriver(b.sim, { heroMax: () => b.run.heroMax && b.run.startWave > 1 });
    const hb = place(b.sim); console.log('hb', !!hb, b.req.sandbox, b.sim.cash, b.sim.towers?.length);
    expect(hb.level).toBe(1);
    let steps = 0; while (hb.xp < heroXpCap(hb.def.hero)) { gainHeroXp(b.sim, hb, 137, 'damage'); steps++; }
    const sa = snap(ha), sb = snap(hb); const diff = [];
    for (const k of new Set([...Object.keys(sa), ...Object.keys(sb)])) if (sa[k] !== sb[k]) diff.push(`${k}: late ${sa[k]} normal ${sb[k]}`);
    console.log('hero', ha.level, ha.mastery, hb.level, hb.mastery, 'steps', steps, 'diffs', diff);
    expect(ha.level).toBe(25); expect(ha.mastery).toBe(hb.mastery);
    da.destroy(); db.destroy();
    expect(Object.hasOwn(a.sim, 'placeTower')).toBe(false);
    // a normal campaign battle in the same module session
    const p = unlocked(); const before = JSON.stringify(p);
    const n = mk({}, p);
    expect(n.req.sandbox).toBeFalsy();
    expect(Object.hasOwn(n.sim, 'placeTower')).toBe(false);
    const hn = place(n.sim);
    expect(hn.level).toBe(1); expect(hn.mastery || 0).toBe(0);
    // sandbox run with resonance on and typical rolls: profile untouched
    const p2 = unlocked(); const b2 = JSON.stringify(p2);
    const s2 = mk({ sandbox: '1', wave: '50', res: 'on', sb: '85-6-5-1-t-lr-best' }, p2);
    const d2 = createSandboxDriver(s2.sim, { heroMax: () => true }); place(s2.sim); d2.destroy();
    expect(JSON.stringify(p2)).toBe(b2);
    expect(JSON.stringify(p)).toBe(before);
  });
});
